# Module 4.1: Event System

## Why Events?

The Observation Layer records **everything** that happens. This enables:
- Real-time dashboard updates
- Debugging and audit trails
- Analytics and insights
- Replay and review

## Event Types

```typescript
// src/types/events.ts

type EventType =
  // Session lifecycle
  | "session_started"
  | "session_ended"

  // Gemini connection
  | "gemini_connected"
  | "gemini_disconnected"

  // Transcriptions
  | "user_transcription"      // User speech fragment
  | "ai_transcription"        // AI speech fragment

  // Tool calls
  | "tool_call_received"      // Gemini emitted a tool call
  | "tool_call_routed_local"  // Handled by vertical
  | "tool_call_routed_remote" // Sent to Agent
  | "tool_call_completed"     // Tool finished successfully
  | "tool_call_failed"        // Tool errored
  | "tool_call_cancelled"     // Tool was cancelled

  // Walkthrough
  | "walkthrough_started"
  | "walkthrough_ended"

  // Issues
  | "issue_flagged"           // Flag captured (includes photo URL)

  // Reports
  | "report_generated"        // CSV report created

  // Photos
  | "photo_captured"          // Manual photo taken
```

## Swift: EventClient

```swift
// Simplified from EventStore/EventClient.swift

actor EventClient {
    private var queue: [Event] = []
    private let maxBatchSize = 20
    private let flushInterval: TimeInterval = 5.0

    private let serverURL: URL
    private var flushTask: Task<Void, Never>?

    init(serverURL: URL) {
        self.serverURL = serverURL
        startAutoFlush()
    }

    func emit(_ type: EventType, payload: [String: Any] = [:]) {
        let event = Event(
            id: UUID().uuidString,
            sessionId: SessionManager.shared.currentSessionId,
            type: type,
            payload: payload,
            timestamp: Date()
        )

        queue.append(event)

        if queue.count >= maxBatchSize {
            Task { await flush() }
        }
    }

    func flush() async {
        guard !queue.isEmpty else { return }

        let batch = Array(queue.prefix(maxBatchSize))
        queue.removeFirst(min(maxBatchSize, queue.count))

        do {
            try await upload(batch)
        } catch {
            // Re-queue failed events
            queue.insert(contentsOf: batch, at: 0)
            // Also persist to disk for offline resilience
            persistToDisk(batch)
        }
    }

    private func upload(_ events: [Event]) async throws {
        var request = URLRequest(url: serverURL)
        request.httpMethod = "POST"
        request.setValue("application/json", forHTTPHeaderField: "Content-Type")
        request.httpBody = try JSONEncoder().encode(events)

        let (_, response) = try await URLSession.shared.data(for: request)
        guard let httpResponse = response as? HTTPURLResponse,
              httpResponse.statusCode == 200 else {
            throw EventUploadError.serverError
        }
    }

    private func startAutoFlush() {
        flushTask = Task {
            while !Task.isCancelled {
                try? await Task.sleep(nanoseconds: UInt64(flushInterval * 1_000_000_000))
                await flush()
            }
        }
    }
}
```

## Web: API Route

```typescript
// src/app/api/events/route.ts

import { NextRequest, NextResponse } from "next/server";
import { supabase } from "@/lib/supabase";

export async function POST(request: NextRequest) {
  try {
    const events = await request.json();

    // Validate
    if (!Array.isArray(events)) {
      return NextResponse.json(
        { error: "Events must be an array" },
        { status: 400 }
      );
    }

    // Insert to Supabase
    const { error } = await supabase
      .from("events")
      .insert(events.map(e => ({
        id: e.id,
        session_id: e.sessionId,
        type: e.type,
        payload: e.payload,
        timestamp: e.timestamp
      })));

    if (error) throw error;

    return NextResponse.json({ success: true });
  } catch (error) {
    console.error("Event ingestion error:", error);
    return NextResponse.json(
      { error: "Internal server error" },
      { status: 500 }
    );
  }
}
```
